1//! Ephemeral VM launch flow for macOS using vfkit + NBD EROFS plugin.
2//!
3//! Boot flow (fully diskless):
4//! 1. Mount container image overlay (`podman image mount`)
5//! 2. Start nbdkit with erofs plugin (dynamically generates GPT + ESP + EROFS)
6//! 3. Launch vfkit with EFI boot via NBD + virtio-net (gvproxy)
7//! 4. Wait for SSH and execute commands
8//!
9//! Common helpers (gvproxy, SSH, vfkit detection) are pub for reuse by vfkit/ module.
10

23/// Base directory for ephemeral VM state on macOS host.
24pub fn ephemeral_base_dir() -> std::path::PathBuf {
25    dirs::home_dir()
26        .unwrap_or_else(|| std::path::PathBuf::from("/tmp"))
27        .join(".local/share/bcvk/ephemeral")
28}

32/// Metadata for a running ephemeral VM, persisted as JSON for `ps` and `ssh`.
33#[derive(serde::Serialize, serde::Deserialize, Debug, Clone)]
34#[allow(dead_code)]
35pub struct EphemeralVmMetadata {
36    /// VM name used as identifier for resource isolation.
37    pub name: String,
38    /// Container image reference used to boot the VM.
39    pub image: String,
40    /// PID of the vfkit process.
41    pub pid: u32,
42    /// PID of the gvproxy network proxy process.
43    pub gvproxy_pid: u32,
44    /// Host-side SSH port forwarded to the VM.
45    pub ssh_port: u16,
46    /// Path to the SSH private key for this VM.
47    pub ssh_key: String,
48    /// Path to the serial console log file.
49    pub serial_log: String,
50    /// Path to the vfkit process log file.
51    pub log_path: Option<String>,
52    /// ISO 8601 timestamp when the VM was created.
53    pub created: String,
54    /// Name of the nbdkit podman container serving the rootfs.
55    #[serde(default)]
56    pub nbd_container: Option<String>,
57    /// NBD port allocated for this VM's rootfs.
58    #[serde(default)]
59    pub nbd_port: Option<u16>,
60}
61
62#[allow(dead_code)]
63impl EphemeralVmMetadata {
64    /// Return the directory path for ephemeral VM metadata files.
65    pub fn vms_dir() -> std::path::PathBuf {
66        ephemeral_base_dir().join("vms")
67    }
68
69    /// Save metadata to a JSON file in the VMs directory.
70    pub fn save(&self) -> Result<()> {
71        let dir = Self::vms_dir();
72        fs::create_dir_all(&dir)?;
73        let path = dir.join(format!("{}.json", self.name));
74        fs::write(&path, serde_json::to_string_pretty(self)?)?;
75        Ok(())
76    }
77
78    /// Remove metadata file for the named VM.
79    pub fn remove(name: &str) {
80        let path = Self::vms_dir().join(format!("{}.json", name));
81        let _ = fs::remove_file(path);
82    }
83
84    /// Load metadata for the named VM from its JSON file.
85    pub fn load(name: &str) -> Result<Self> {
86        let path = Self::vms_dir().join(format!("{}.json", name));
87        let data = fs::read_to_string(&path)?;
88        Ok(serde_json::from_str(&data)?)
89    }
90
91    /// List all ephemeral VM metadata from the VMs directory.
92    pub fn list_all() -> Result<Vec<Self>> {
93        let dir = Self::vms_dir();
94        if !dir.exists() {
95            return Ok(Vec::new());
96        }
97        let mut vms = Vec::new();
98        for entry in fs::read_dir(&dir)? {
99            let path = entry?.path();
100            if path.extension().and_then(|e| e.to_str()) != Some("json") {
101                continue;
102            }
103            if let Ok(data) = fs::read_to_string(&path) {
104                if let Ok(meta) = serde_json::from_str::<Self>(&data) {
105                    vms.push(meta);
106                }
107            }
108        }
109        Ok(vms)
110    }
111
112    /// Check if the VM process is still alive via kill(pid, 0).
113    pub fn is_alive(&self) -> bool {
114        rustix::process::test_kill_process(rustix::process::Pid::from_raw(self.pid as i32).unwrap())
115            .is_ok()
116    }
117}

119/// Options for launching an ephemeral VM via vfkit.
120#[derive(clap::Parser, Debug)]
121pub struct RunEphemeralOpts {
122    /// Container image to boot
123    pub image: String,
124    /// Number of vCPUs
125    #[clap(long)]
126    pub vcpus: Option<u32>,
127    /// Memory size (e.g. "4G", "2048M", or plain number for MB)
128    #[clap(long, default_value = "4G")]
129    pub memory: String,
130    /// Generate a temporary SSH key pair for VM access
131    #[clap(long = "ssh-keygen", short = 'K')]
132    pub ssh_keygen: bool,
133    /// Command(s) to execute via SSH after boot
134    #[clap(long)]
135    pub execute: Vec<String>,
136    /// VM name for identification and resource isolation
137    #[clap(long)]
138    pub name: Option<String>,
139    /// Additional kernel command line arguments
140    #[clap(long = "karg")]
141    pub kernel_args: Vec<String>,
142    /// Display VM console in GUI window
143    #[clap(long)]
144    pub gui: bool,
145    /// Run in background
146    #[clap(long, short = 'd')]
147    pub detach: bool,
148    /// Enable debug mode (reserved for future use)
149    #[clap(long)]
150    pub debug: bool,
151}

153fn default_vcpus() -> u32 {
154    std::thread::available_parallelism()
155        .map(|n| n.get() as u32)
156        .unwrap_or(2)
157}

159/// Parse memory specification string (e.g. "4G", "2048M") to megabytes.
160pub fn parse_memory_to_mb(s: &str) -> Result<u32> {
161    let s = s.trim();
162    if let Some(n) = s.strip_suffix('G').or_else(|| s.strip_suffix('g')) {
163        Ok((n.parse::<f64>()? * 1024.0) as u32)
164    } else if let Some(n) = s.strip_suffix('M').or_else(|| s.strip_suffix('m')) {
165        Ok(n.parse::<f64>()? as u32)
166    } else {
167        Ok(s.parse::<u32>()?)
168    }
169}

171// --- RAII cleanup guard ---

173struct VmCleanup {
174    vfkit_pid: u32,
175    gvproxy_pid: u32,
176    nbd_container: Option<String>,
177    image: String,
178    vm_name: String,
179}
180
181impl Drop for VmCleanup {
182    fn drop(&mut self) {
183        tracing::debug!("cleaning up VM processes...");
184        if let Some(ref name) = self.nbd_container {
185            crate::nbdkit_macos::stop_nbdkit_container(name);
186        }
187        if let Err(e) = rustix::process::kill_process(
188            rustix::process::Pid::from_raw(self.vfkit_pid as i32).unwrap(),
189            rustix::process::Signal::TERM,
190        ) {
191            tracing::warn!("failed to kill vfkit (PID {}): {}", self.vfkit_pid, e);
192        }
193        if let Err(e) = rustix::process::kill_process(
194            rustix::process::Pid::from_raw(self.gvproxy_pid as i32).unwrap(),
195            rustix::process::Signal::TERM,
196        ) {
197            tracing::warn!("failed to kill gvproxy (PID {}): {}", self.gvproxy_pid, e);
198        }
199        // Release container image overlay mount
200        if let Ok(machine) = detect_machine_name() {
201            let _ = Command::new("podman")
202                .args([
203                    "machine",
204                    "ssh",
205                    &machine,
206                    "--",
207                    "podman",
208                    "image",
209                    "umount",
210                    &self.image,
211                ])
212                .stdout(Stdio::null())
213                .stderr(Stdio::null())
214                .status();
215        }
216        EphemeralVmMetadata::remove(&self.vm_name);
217    }
218}

222/// Run an ephemeral VM from a container image using vfkit + EROFS over NBD.
223pub fn run(opts: RunEphemeralOpts) -> Result<()> {
224    if opts.gui && opts.detach {
225        bail!("--gui and --detach cannot be used together (GUI requires foreground process)");
226    }
227
228    if opts.detach {
229        return run_detached(&opts);
230    }
231
232    let vfkit_bin = find_vfkit()?;
233    info!(image = %opts.image, "starting ephemeral VM on macOS (vfkit + EROFS)");
234
235    let cache_base = ephemeral_base_dir();
236    fs::create_dir_all(&cache_base)?;
237
238    let machine = detect_machine_name()?;
239    let rootful = is_machine_rootful(&machine);
240    debug!(
241        "podman machine '{}' ({})",
242        machine,
243        if rootful { "rootful" } else { "rootless" }
244    );
245    let digest = ensure_image_and_get_digest(&opts.image)?;
246    let digest_short = &digest[..16.min(digest.len())];
247    info!("image digest: {}...", digest_short);
248
249    let vm_name = opts
250        .name
251        .clone()
252        .unwrap_or_else(|| format!("ephemeral-{}", &digest_short[..8]));
253    let ssh_key_path = cache_base.join(format!("{}-key", vm_name));
254
255    fs::create_dir_all(&cache_base)?;
256
257    // Generate SSH keypair on macOS host
258    let mut ssh_pubkey = String::new();
259    if opts.ssh_keygen || !opts.execute.is_empty() {
260        info!("generating SSH keypair...");
261        let _ = fs::remove_file(&ssh_key_path);
262        let _ = fs::remove_file(ssh_key_path.with_extension("pub"));
263        let status = Command::new("ssh-keygen")
264            .args([
265                "-t",
266                "ed25519",
267                "-f",
268                &ssh_key_path.to_string_lossy(),
269                "-N",
270                "",
271                "-q",
272            ])
273            .status()?;
274        if !status.success() {
275            bail!("ssh-keygen failed");
276        }
277        ssh_pubkey = fs::read_to_string(ssh_key_path.with_extension("pub"))?
278            .trim()
279            .to_string();
280    }
281
282    let mut cmdline_parts: Vec<&str> = vec![
283        "root=/dev/vda2",
284        "ro",
285        "rootfstype=erofs",
286        "console=tty0",
287        "console=hvc0",
288        "loglevel=4",
289        "selinux=0",
290        "net.ifnames=0",
291        "systemd.journald.storage=volatile",
292    ];
293    let user_args: Vec<&str> = opts.kernel_args.iter().map(|s| s.as_str()).collect();
294    cmdline_parts.extend(&user_args);
295    let cmdline = cmdline_parts.join(" ");
296
297    // Get container image merged overlay path
298    let merged_path = crate::nbdkit_macos::get_merged_path(&machine, rootful, &opts.image)?;
299    info!("overlay merged: {}", merged_path);
300
301    // Start nbdkit with erofs plugin (dynamic EROFS + ESP + GPT from overlay dir)
302    let nbd_port = crate::nbdkit_macos::find_available_nbd_port();
303    let nbd_container_name = crate::nbdkit_macos::start_nbdkit_erofs_plugin(
304        &machine,
305        &merged_path,
306        &cmdline,
307        &ssh_pubkey,
308        nbd_port,
309        &vm_name,
310    )?;
311    std::thread::sleep(Duration::from_millis(500));
312    info!("nbdkit ready on port {}", nbd_port);
313
314    // gvproxy + vfkit (EFI boot)
315    let gvproxy_sock = cache_base.join(format!("{}-gvproxy.sock", vm_name));
316    let services_sock = cache_base.join(format!("{}-gvproxy-svc.sock", vm_name));
317    let gvproxy_sock_str = gvproxy_sock.to_string_lossy().to_string();
318    let services_sock_str = services_sock.to_string_lossy().to_string();
319    info!("starting gvproxy...");
320    let mut gvproxy_child = start_gvproxy(&gvproxy_sock_str, &services_sock_str)?;
321
322    let mac = generate_mac();
323    let mac_str = format!(
324        "{:02x}:{:02x}:{:02x}:{:02x}:{:02x}:{:02x}",
325        mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]
326    );
327
328    let efi_var_store = cache_base.join(format!("{}-efi-vars", vm_name));
329    let bootloader_arg = format!("efi,variable-store={},create", efi_var_store.display());
330
331    let vcpus = opts.vcpus.unwrap_or_else(default_vcpus);
332    let memory_mb = parse_memory_to_mb(&opts.memory)?;
333
334    let mut vfkit_args = vec![
335        "--cpus".to_string(),
336        vcpus.to_string(),
337        "--memory".to_string(),
338        memory_mb.to_string(),
339        "--bootloader".to_string(),
340        bootloader_arg,
341        "--device".to_string(),
342        format!(
343            "nbd,uri=nbd://127.0.0.1:{}/,readonly,timeout=5000,deviceId=rootfs",
344            nbd_port
345        ),
346        "--device".to_string(),
347        format!(
348            "virtio-net,unixSocketPath={},mac={}",
349            gvproxy_sock_str, mac_str
350        ),
351        "--device".to_string(),
352        "virtio-rng".to_string(),
353    ];
354
355    let serial_log = cache_base.join(format!("{}-serial.log", vm_name));
356    vfkit_args.extend([
357        "--device".to_string(),
358        format!("virtio-serial,logFilePath={}", serial_log.display()),
359    ]);
360
361    if opts.gui {
362        vfkit_args.push("--gui".to_string());
363    }
364
365    info!("launching vfkit...");
366    let vfkit_log = cache_base.join(format!("{}-vfkit.log", vm_name));
367    let vfkit_log_file = fs::File::create(&vfkit_log)?;
368    let mut vfkit_child = Command::new(&vfkit_bin)
369        .args(&vfkit_args)
370        .stdout(vfkit_log_file.try_clone()?)
371        .stderr(vfkit_log_file)
372        .spawn()
373        .context("failed to start vfkit")?;
374
375    let ssh_port = find_available_ssh_port();
376    debug!("allocated SSH port: {}", ssh_port);
377
378    let metadata = EphemeralVmMetadata {
379        name: vm_name.clone(),
380        image: opts.image.clone(),
381        pid: vfkit_child.id(),
382        gvproxy_pid: gvproxy_child.id(),
383        ssh_port,
384        ssh_key: ssh_key_path.to_string_lossy().to_string(),
385        serial_log: serial_log.to_string_lossy().to_string(),
386        log_path: None,
387        created: chrono::Utc::now().to_rfc3339(),
388        nbd_container: Some(nbd_container_name.clone()),
389        nbd_port: Some(nbd_port),
390    };
391    metadata.save()?;
392
393    let _cleanup = VmCleanup {
394        vfkit_pid: vfkit_child.id(),
395        gvproxy_pid: gvproxy_child.id(),
396        nbd_container: Some(nbd_container_name.clone()),
397        image: opts.image.clone(),
398        vm_name: vm_name.clone(),
399    };
400
401    if opts.ssh_keygen || !opts.execute.is_empty() {
402        info!("setting up SSH port forwarding...");
403        for attempt in 0..15u32 {
404            match expose_ssh_port(&services_sock_str, "192.168.127.2", ssh_port) {
405                Ok(_) => {
406                    info!("SSH port {} forwarded", ssh_port);
407                    break;
408                }
409                Err(e) if attempt < 14 => {
410                    debug!("SSH port forward attempt {}: {}", attempt, e);
411                    let backoff = 200 * 2u64.pow(attempt.min(4));
412                    std::thread::sleep(Duration::from_millis(backoff));
413                }
414                Err(e) => bail!("SSH port forward failed: {}", e),
415            }
416        }
417
418        wait_for_ssh(ssh_port, &ssh_key_path, "root")?;
419
420        if !opts.execute.is_empty() {
421            for cmd_str in &opts.execute {
422                info!("executing: {}", cmd_str);
423                let status = run_ssh_command(ssh_port, &ssh_key_path, "root", cmd_str)?;
424                if !status.success() {
425                    bail!("command failed: {}", status);
426                }
427            }
428            return Ok(());
429        }
430
431        info!(
432            "SSH ready: ssh -p {} -i {} root@localhost",
433            ssh_port,
434            ssh_key_path.display()
435        );
436
437        use std::io::IsTerminal;
438        if std::io::stdin().is_terminal() {
439            let status = run_ssh_interactive(ssh_port, &ssh_key_path, "root")?;
440            let exit_code = status.code().unwrap_or(1);
441            drop(_cleanup);
442            std::process::exit(exit_code);
443        }
444    }
445
446    // No SSH: wait for vfkit to exit (GUI window closed or VM shutdown)
447    std::mem::forget(_cleanup);
448    let status = vfkit_child.wait()?;
449    info!("vfkit exited: {}", status);
450    crate::nbdkit_macos::stop_nbdkit_container(&nbd_container_name);
451    if let Err(e) = gvproxy_child.kill() {
452        tracing::debug!("failed to kill gvproxy: {}", e);
453    }
454    // Release container image overlay mount
455    let _ = Command::new("podman")
456        .args([
457            "machine",
458            "ssh",
459            &machine,
460            "--",
461            "podman",
462            "image",
463            "umount",
464            &opts.image,
465        ])
466        .stdout(Stdio::null())
467        .stderr(Stdio::null())
468        .status();
469    EphemeralVmMetadata::remove(&vm_name);
470    Ok(())
471}

473fn run_detached(opts: &RunEphemeralOpts) -> Result<()> {
474    let cache_base = ephemeral_base_dir();
475    fs::create_dir_all(&cache_base)?;
476    let digest = ensure_image_and_get_digest(&opts.image)?;
477    let digest_short = &digest[..16.min(digest.len())];
478    let vm_name = opts
479        .name
480        .clone()
481        .unwrap_or_else(|| format!("ephemeral-{}", &digest_short[..8]));
482    let log_path = cache_base.join(format!("bcvk-{}.log", vm_name));
483    let log_file = fs::File::create(&log_path)?;
484
485    let exe = std::env::current_exe()?;
486    let mut args: Vec<String> = std::env::args()
487        .skip(1)
488        .filter(|a| a != "--detach" && a != "-d")
489        .collect();
490    if !args.contains(&"-K".to_string()) && !args.contains(&"--ssh-keygen".to_string()) {
491        args.insert(args.len() - 1, "-K".to_string());
492    }
493    if opts.name.is_none() {
494        args.insert(args.len() - 1, "--name".to_string());
495        args.insert(args.len() - 1, vm_name.clone());
496    }
497
498    let child = Command::new(exe)
499        .args(&args)
500        .stdin(Stdio::null())
501        .stdout(log_file.try_clone()?)
502        .stderr(log_file)
503        .spawn()?;
504
505    let metadata = EphemeralVmMetadata {
506        name: vm_name.clone(),
507        image: opts.image.clone(),
508        pid: child.id(),
509        gvproxy_pid: 0,
510        ssh_port: 0,
511        ssh_key: cache_base
512            .join(format!("{}-key", vm_name))
513            .to_string_lossy()
514            .to_string(),
515        serial_log: String::new(),
516        log_path: Some(log_path.to_string_lossy().to_string()),
517        created: chrono::Utc::now().to_rfc3339(),
518        nbd_container: None,
519        nbd_port: None,
520    };
521    metadata.save()?;
522    println!("{}", vm_name);
523    Ok(())
524}

528/// Detect the name of the running podman machine.
529pub fn detect_machine_name() -> Result<String> {
530    let output = Command::new("podman")
531        .args(["machine", "info", "--format", "{{.Host.CurrentMachine}}"])
532        .output()?;
533    let name = String::from_utf8_lossy(&output.stdout).trim().to_string();
534    if name.is_empty() {
535        bail!("no podman machine is running");
536    }
537    Ok(name)
538}

540fn ensure_image_and_get_digest(image: &str) -> Result<String> {
541    let status = Command::new("podman")
542        .args(["image", "exists", image])
543        .stdout(Stdio::null())
544        .stderr(Stdio::null())
545        .status()?;
546    if !status.success() {
547        info!("pulling image {}...", image);
548        if !Command::new("podman")
549            .args(["pull", image])
550            .status()?
551            .success()
552        {
553            bail!("failed to pull image: {}", image);
554        }
555    }
556    let output = Command::new("podman")
557        .args(["image", "inspect", "--format", "{{.Digest}}", image])
558        .output()?;
559    let digest = String::from_utf8_lossy(&output.stdout).trim().to_string();
560    Ok(digest.trim_start_matches("sha256:").to_string())
561}

563fn is_machine_rootful(machine: &str) -> bool {
564    Command::new("podman")
565        .args(["machine", "ssh", machine, "id", "-u"])
566        .output()
567        .map(|o| String::from_utf8_lossy(&o.stdout).trim() == "0")
568        .unwrap_or(false)

584/// Find the vfkit binary, checking PATH and Podman PKG location.
585pub fn find_vfkit() -> Result<String> {
586    if let Ok(path) = which::which("vfkit") {
587        return Ok(path.to_string_lossy().to_string());
588    }
589    let podman_path = "/opt/podman/bin/vfkit";
590    if Path::new(podman_path).exists() {
591        return Ok(podman_path.to_string());
592    }
593    bail!("vfkit not found. Install: brew install vfkit")
594}

604/// Find the gvproxy binary, checking PATH and Podman installation paths.
605fn find_gvproxy() -> Result<String> {
606    if let Ok(path) = which::which("gvproxy") {
607        return Ok(path.to_string_lossy().to_string());
608    }
609    for candidate in [
610        "/opt/homebrew/opt/podman/libexec/podman/gvproxy",
611        "/opt/podman/bin/gvproxy",
612    ] {
613        if Path::new(candidate).exists() {
614            return Ok(candidate.to_string());
615        }
616    }
617    bail!("gvproxy not found. Ensure Podman is installed (brew install podman)")
618}

620/// Start a gvproxy instance with the given socket paths.
621pub fn start_gvproxy(gvproxy_sock: &str, services_sock: &str) -> Result<std::process::Child> {
622    let gvproxy_bin = find_gvproxy()?;
623    let _ = fs::remove_file(gvproxy_sock);
624    let _ = fs::remove_file(services_sock);
625    let child = Command::new(&gvproxy_bin)
626        .args([
627            "-listen-vfkit",
628            &format!("unixgram://{}", gvproxy_sock),
629            "-ssh-port",
630            "-1",
631            "-services",
632            &format!("unix://{}", services_sock),
633        ])
634        .stdout(Stdio::null())
635        .stderr(Stdio::null())
636        .spawn()
637        .context("failed to start gvproxy. Ensure gvproxy is installed (included in Podman)")?;
638    for _ in 0..50 {
639        if Path::new(gvproxy_sock).exists() {
640            break;
641        }
642        std::thread::sleep(Duration::from_millis(100));
643    }
644    if !Path::new(gvproxy_sock).exists() {
645        bail!("gvproxy socket did not appear");
646    }
647    Ok(child)
648}

650/// Expose SSH port forwarding via gvproxy's HTTP API.
651pub fn expose_ssh_port(services_sock: &str, vm_ip: &str, host_port: u16) -> Result<()> {
652    let body = format!(
653        r#"{{"local":":{}","remote":"{}:22","protocol":"tcp"}}"#,
654        host_port, vm_ip
655    );
656    let mut stream = UnixStream::connect(services_sock)?;
657    let request = format!(
658        "POST /services/forwarder/expose HTTP/1.1\r\nHost: unix\r\n\
659         Content-Type: application/json\r\nContent-Length: {}\r\n\r\n{}",
660        body.len(),
661        body
662    );
663    std::io::Write::write_all(&mut stream, request.as_bytes())?;
664    std::io::Write::flush(&mut stream)?;
665    let mut response = vec![0u8; 1024];
666    let _ = std::io::Read::read(&mut stream, &mut response);
667    let response_str = String::from_utf8_lossy(&response);
668    if !response_str.contains("200") {
669        bail!(
670            "gvproxy expose failed: {}",
671            response_str.trim_end_matches('\0')
672        );
673    }
674    Ok(())
675}

677const SSH_TIMEOUT: Duration = Duration::from_secs(240);
678
679/// Find an available TCP port for SSH forwarding in range 2222-3000.
680pub fn find_available_ssh_port() -> u16 {
681    use rand::Rng;
682    let mut rng = rand::rng();
683    const PORT_RANGE_START: u16 = 2222;
684    const PORT_RANGE_END: u16 = 3000;
685    for _ in 0..100 {
686        let port = rng.random_range(PORT_RANGE_START..PORT_RANGE_END);
687        if std::net::TcpListener::bind(("127.0.0.1", port)).is_ok() {
688            return port;
689        }
690    }
691    for port in PORT_RANGE_START..PORT_RANGE_END {
692        if std::net::TcpListener::bind(("127.0.0.1", port)).is_ok() {
693            return port;
694        }
695    }
696    PORT_RANGE_START
697}

699/// Wait for SSH connectivity with exponential backoff (240s timeout).
700pub fn wait_for_ssh(port: u16, key_path: &Path, user: &str) -> Result<()> {
701    use crate::ssh_options::CommonSshOptions;
702    let ssh_opts = CommonSshOptions::default();
703    let user_host = format!("{}@localhost", user);
704    info!("waiting for SSH on port {} ({}@localhost)...", port, user);
705    let start = std::time::Instant::now();
706    let mut attempt = 0u32;
707    loop {
708        if start.elapsed() > SSH_TIMEOUT {
709            bail!("SSH connection timeout ({}s)", SSH_TIMEOUT.as_secs());
710        }
711        let mut cmd = Command::new("ssh");
712        cmd.args(["-p", &port.to_string(), "-i", &key_path.to_string_lossy()]);
713        ssh_opts.apply_to_command(&mut cmd);
714        cmd.args(["-o", "BatchMode=yes", &user_host, "true"]);
715        let status = cmd.stdout(Stdio::null()).stderr(Stdio::null()).status();
716        if let Ok(s) = status {
717            if s.success() {
718                info!("SSH connected after {}s", start.elapsed().as_secs());
719                return Ok(());
720            }
721        }
722        let backoff = if attempt < 2 {
723            500
724        } else if attempt < 4 {
725            1000
726        } else {
727            2000
728        };
729        std::thread::sleep(Duration::from_millis(backoff));
730        attempt += 1;
731    }
732}

735pub fn run_ssh_command(
736    port: u16,
737    key_path: &Path,
738    user: &str,
739    command: &str,
740) -> Result<std::process::ExitStatus> {
741    use crate::ssh_options::CommonSshOptions;

743    let user_host = format!("{}@localhost", user);

746    ssh_opts.apply_to_command(&mut cmd);
747    cmd.args(["-o", "BatchMode=yes", &user_host, command]);
748    cmd.stdin(Stdio::inherit())
749        .stdout(Stdio::inherit())
750        .stderr(Stdio::inherit())
751        .status()
752        .map_err(|e| eyre!("ssh failed: {}", e))
