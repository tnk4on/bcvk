1//! Helper functions for interacting with qemu-img
2
3// On non-Linux, this module is unused as it's for qemu-img operations
4#![cfg_attr(not(target_os = "linux"), allow(dead_code))]
5
6use camino::Utf8Path;
7use color_eyre::{eyre::Context, Result};
8use serde::Deserialize;
9use std::process::Command;
10
11/// Information returned by `qemu-img info --output=json`
12#[derive(Debug, Deserialize)]
13#[serde(rename_all = "kebab-case")]
14#[allow(dead_code)]
15pub struct QemuImgInfo {
16    /// Virtual size of the disk image in bytes
17    pub virtual_size: u64,
18    /// Path to the disk image file
19    pub filename: String,
20    /// Image format (e.g., "qcow2", "raw")
21    pub format: String,
22    /// Actual size on disk in bytes (if available)
23    pub actual_size: Option<u64>,
24    /// Cluster size in bytes (for formats like qcow2)
25    pub cluster_size: Option<u64>,
26    /// Backing file name (if this is a snapshot)
27    pub backing_filename: Option<String>,
28    /// Full path to backing file (if this is a snapshot)
29    pub full_backing_filename: Option<String>,
30    /// Whether the image is marked as dirty
31    pub dirty_flag: Option<bool>,
32}
33
34/// Run `qemu-img info --force-share --output=json` on a disk image
35///
36/// The `--force-share` flag allows reading disk info even when the image
37/// is locked by a running VM.
38pub fn info(path: &Utf8Path) -> Result<QemuImgInfo> {
39    let output = Command::new("qemu-img")
40        .args(["info", "--force-share", "--output=json", path.as_str()])
41        .output()
42        .with_context(|| format!("Failed to run qemu-img info on {:?}", path))?;
43
44    if !output.status.success() {
45        let stderr = String::from_utf8_lossy(&output.stderr);
46        return Err(color_eyre::eyre::eyre!(
47            "qemu-img info failed for {:?}: {}",
48            path,
49            stderr
50        ));
51    }
52
53    serde_json::from_slice(&output.stdout)
54        .with_context(|| format!("Failed to parse qemu-img info JSON for {:?}", path))
55}
