17/// Options for `vm run`.
18#[derive(Parser, Debug)]
19pub struct VmRunOpts {
20    /// Disk image path (.raw)
21    pub disk: String,
22    /// VM name for identification
23    #[clap(long)]
24    pub name: Option<String>,
25    /// Number of vCPUs
26    #[clap(long)]
27    pub vcpus: Option<u32>,
28    /// Memory size (e.g. "4G", "2048M", or plain number for MB)
29    #[clap(long, default_value = "4G")]
30    pub memory: String,
31    /// Path to an existing SSH private key
32    #[clap(long)]
33    pub ssh_key: Option<String>,
34    /// SSH username (default: root)
35    #[clap(long, default_value = "root")]
36    pub ssh_user: String,
37    /// SSH port (default: auto-allocate)
38    #[clap(long)]
39    pub ssh_port: Option<u16>,
40    /// Display VM console in GUI window
41    #[clap(long)]
42    pub gui: bool,
43}

45/// Create and launch a persistent VM from a disk image via vfkit + EFI.
46pub fn run(opts: VmRunOpts) -> Result<()> {
47    let vfkit_bin = find_vfkit()?;
48
49    if !Path::new(&opts.disk).exists() {
50        bail!("disk image not found: {}", opts.disk);
51    }
52    clear_xattr(Path::new(&opts.disk));
53
54    let ssh_key_path = match &opts.ssh_key {
55        Some(p) => p.clone(),
56        None => find_ssh_key()?,
57    };
58    if !Path::new(&ssh_key_path).exists() {
59        bail!(
60            "SSH key not found: {}. Specify with --ssh-key",
61            ssh_key_path
62        );
63    }
64
65    let vm_name = opts.name.clone().unwrap_or_else(|| {
66        Path::new(&opts.disk)
67            .file_stem()
68            .and_then(|s| s.to_str())
69            .unwrap_or("vm")
70            .to_string()
71    });
72
73    let vms_dir = VmMetadata::vms_dir();
74    fs::create_dir_all(&vms_dir)?;
75
76    let efi_store = vms_dir.join(format!("{}-efi-vars", vm_name));
77    let serial_log = vms_dir.join(format!("{}-serial.log", vm_name));
78    let gvproxy_sock = vms_dir.join(format!("{}-gvproxy.sock", vm_name));
79    let services_sock = vms_dir.join(format!("{}-gvproxy-svc.sock", vm_name));
80
81    let gvproxy_sock_str = gvproxy_sock.to_string_lossy().to_string();
82    let services_sock_str = services_sock.to_string_lossy().to_string();
83
84    info!("starting gvproxy...");
85    let gvproxy_child = start_gvproxy(&gvproxy_sock_str, &services_sock_str)?;
86
87    let mac = generate_mac();
88    let mac_str = format!(
89        "{:02x}:{:02x}:{:02x}:{:02x}:{:02x}:{:02x}",
90        mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]
91    );
92
93    let vcpus = opts.vcpus.unwrap_or(2);
94    let memory_mb = crate::run_ephemeral_macos::parse_memory_to_mb(&opts.memory)?;
95
96    let mut vfkit_args = vec![
97        "--cpus".to_string(),
98        vcpus.to_string(),
99        "--memory".to_string(),
100        memory_mb.to_string(),
101        "--bootloader".to_string(),
102        format!("efi,variable-store={},create", efi_store.display()),
103        "--device".to_string(),
104        format!("virtio-blk,path={}", opts.disk),
105        "--device".to_string(),
106        format!(
107            "virtio-net,unixSocketPath={},mac={}",
108            gvproxy_sock_str, mac_str
109        ),
110        "--device".to_string(),
111        format!("virtio-serial,logFilePath={}", serial_log.display()),
112        "--device".to_string(),
113        "virtio-rng".to_string(),
114    ];
115    if opts.gui {
116        vfkit_args.push("--gui".to_string());
117    }
118
119    info!("launching vfkit (EFI boot)...");
120    let vfkit_child = Command::new(&vfkit_bin)
121        .args(&vfkit_args)
122        .stdout(Stdio::null())
123        .stderr(Stdio::null())
124        .spawn()?;
125
126    let ssh_port = opts.ssh_port.unwrap_or_else(find_available_ssh_port);
127    info!("SSH port: {}", ssh_port);
128
129    info!("setting up SSH port forwarding...");
130    for attempt in 0..15u32 {
131        match expose_ssh_port(&services_sock_str, "192.168.127.2", ssh_port) {
132            Ok(_) => {
133                info!("SSH port {} forwarded", ssh_port);
134                break;
135            }
136            Err(e) if attempt < 14 => {
137                tracing::debug!("SSH port forward attempt {}: {}", attempt, e);
138                let backoff = 200 * 2u64.pow(attempt.min(4));
139                std::thread::sleep(std::time::Duration::from_millis(backoff));
140            }
141            Err(e) => bail!("SSH port forward failed: {}", e),
142        }
143    }
144
145    let key_path = std::path::Path::new(&ssh_key_path);
146    wait_for_ssh(ssh_port, key_path, &opts.ssh_user)?;
147
148    let metadata = VmMetadata {
149        name: vm_name.clone(),
150        disk_image: opts.disk.clone(),
151        vfkit_pid: vfkit_child.id(),
152        gvproxy_pid: gvproxy_child.id(),
153        ssh_port,
154        ssh_key: ssh_key_path.clone(),
155        ssh_user: opts.ssh_user.clone(),
156        cpus: vcpus,
157        memory: memory_mb,
158        efi_store: efi_store.to_string_lossy().to_string(),
159        serial_log: serial_log.to_string_lossy().to_string(),
160        gui: opts.gui,
161        created: chrono::Utc::now().to_rfc3339(),
162        state: "running".to_string(),
163    };
164    metadata.save()?;
165
166    println!("VM '{}' is running", vm_name);
167    println!(
168        "  ssh -p {} -i {} {}@localhost",
169        ssh_port, ssh_key_path, opts.ssh_user
170    );
171    println!();
172    println!("To connect:  bcvk vm ssh {}", vm_name);
173    println!("To stop:     bcvk vm stop {}", vm_name);
174
175    Ok(())
176}

178fn find_ssh_key() -> Result<String> {
179    let home = dirs::home_dir()
180        .ok_or_else(|| color_eyre::eyre::eyre!("cannot determine home directory"))?;
181    for name in &["id_ed25519", "id_rsa"] {
182        let path = home.join(".ssh").join(name);
183        if path.exists() {
184            return Ok(path.to_string_lossy().to_string());
185        }
186    }
187    bail!("no SSH key found in ~/.ssh/. Generate with: ssh-keygen -t ed25519")
188}
