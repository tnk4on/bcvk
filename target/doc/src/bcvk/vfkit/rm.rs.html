1//! vm rm — Remove a persistent VM and its metadata.
2
3use std::fs;
4
5use clap::Parser;
6use color_eyre::Result;
7use tracing::info;
8
9use super::VmMetadata;
10
11/// Options for `vm rm`.
12#[derive(Parser, Debug)]
13pub struct VmRmOpts {
14    /// VM name
15    pub name: String,
16    /// Force removal even if running
17    #[clap(short, long)]
18    pub force: bool,
19}
20
21/// Remove a persistent VM, optionally force-killing it.
22pub fn run(opts: VmRmOpts) -> Result<()> {
23    let meta = VmMetadata::load(&opts.name)?;
24
25    if meta.is_alive() {
26        if !opts.force {
27            color_eyre::eyre::bail!(
28                "VM '{}' is running. Stop it first or use --force",
29                opts.name
30            );
31        }
32        info!("force stopping VM '{}'...", opts.name);
33        crate::vfkit::stop::run(&opts.name)?;
34    }
35
36    for path in [&meta.efi_store, &meta.serial_log] {
37        if !path.is_empty() {
38            if let Err(e) = fs::remove_file(path) {
39                if e.kind() != std::io::ErrorKind::NotFound {
40                    tracing::debug!("failed to remove {}: {}", path, e);
41                }
42            }
43        }
44    }
45
46    let vms_dir = VmMetadata::vms_dir();
47    for suffix in ["-gvproxy.sock", "-gvproxy-svc.sock"] {
48        let p = vms_dir.join(format!("{}{}", meta.name, suffix));
49        if let Err(e) = fs::remove_file(&p) {
50            if e.kind() != std::io::ErrorKind::NotFound {
51                tracing::debug!("failed to remove {}: {}", p.display(), e);
52            }
53        }
54    }
55
56    VmMetadata::remove(&opts.name);
57    println!("Removed '{}'", opts.name);
58    Ok(())
59}
