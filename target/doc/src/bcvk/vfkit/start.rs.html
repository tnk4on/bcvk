1//! vm start — Restart a stopped persistent VM.
2
3use std::process::{Command, Stdio};
4
5use clap::Parser;
6use color_eyre::{eyre::bail, Result};
7use tracing::info;
8
9use super::VmMetadata;
10use crate::run_ephemeral_macos::{
11    clear_xattr, expose_ssh_port, find_vfkit, generate_mac, start_gvproxy, wait_for_ssh,
12};
13
14/// Options for `vm start`.
15#[derive(Parser, Debug)]
16pub struct VmStartOpts {
17    /// VM name
18    pub name: String,
19    /// Display VM console in GUI window
20    #[clap(long)]
21    pub gui: bool,
22}
23
24/// Restart a stopped persistent VM by re-launching vfkit.
25pub fn run(opts: VmStartOpts) -> Result<()> {
26    let mut meta = VmMetadata::load(&opts.name)?;
27    if meta.is_alive() {
28        bail!("VM '{}' is already running", opts.name);
29    }
30
31    if !std::path::Path::new(&meta.disk_image).exists() {
32        bail!("disk image not found: {}", meta.disk_image);
33    }
34    clear_xattr(std::path::Path::new(&meta.disk_image));
35
36    let vfkit_bin = find_vfkit()?;
37    let vms_dir = VmMetadata::vms_dir();
38
39    let gvproxy_sock = vms_dir.join(format!("{}-gvproxy.sock", meta.name));
40    let services_sock = vms_dir.join(format!("{}-gvproxy-svc.sock", meta.name));
41    let gvproxy_sock_str = gvproxy_sock.to_string_lossy().to_string();
42    let services_sock_str = services_sock.to_string_lossy().to_string();
43
44    info!("starting gvproxy...");
45    let gvproxy_child = start_gvproxy(&gvproxy_sock_str, &services_sock_str)?;
46
47    let mac = generate_mac();
48    let mac_str = format!(
49        "{:02x}:{:02x}:{:02x}:{:02x}:{:02x}:{:02x}",
50        mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]
51    );
52
53    let gui = opts.gui || meta.gui;
54    let mut vfkit_args = vec![
55        "--cpus".to_string(),
56        meta.cpus.to_string(),
57        "--memory".to_string(),
58        meta.memory.to_string(),
59        "--bootloader".to_string(),
60        format!("efi,variable-store={},create", meta.efi_store),
61        "--device".to_string(),
62        format!("virtio-blk,path={}", meta.disk_image),
63        "--device".to_string(),
64        format!(
65            "virtio-net,unixSocketPath={},mac={}",
66            gvproxy_sock_str, mac_str
67        ),
68        "--device".to_string(),
69        format!("virtio-serial,logFilePath={}", meta.serial_log),
70        "--device".to_string(),
71        "virtio-rng".to_string(),
72    ];
73    if gui {
74        vfkit_args.push("--gui".to_string());
75    }
76
77    info!("launching vfkit (EFI boot)...");
78    let vfkit_child = Command::new(&vfkit_bin)
79        .args(&vfkit_args)
80        .stdout(Stdio::null())
81        .stderr(Stdio::null())
82        .spawn()?;
83
84    info!("setting up SSH port forwarding...");
85    for attempt in 0..15u32 {
86        match expose_ssh_port(&services_sock_str, "192.168.127.2", meta.ssh_port) {
87            Ok(_) => {
88                info!("SSH port {} forwarded", meta.ssh_port);
89                break;
90            }
91            Err(e) if attempt < 14 => {
92                tracing::debug!("SSH port forward attempt {}: {}", attempt, e);
93                let backoff = 200 * 2u64.pow(attempt.min(4));
94                std::thread::sleep(std::time::Duration::from_millis(backoff));
95            }
96            Err(e) => bail!("SSH port forward failed: {}", e),
97        }
98    }
99
100    let key_path = std::path::Path::new(&meta.ssh_key);
101    wait_for_ssh(meta.ssh_port, key_path, &meta.ssh_user)?;
102
103    meta.vfkit_pid = vfkit_child.id();
104    meta.gvproxy_pid = gvproxy_child.id();
105    meta.state = "running".to_string();
106    meta.gui = gui;
107    meta.save()?;
108
109    println!("Started '{}'", meta.name);
110    println!(
111        "  ssh -p {} -i {} {}@localhost",
112        meta.ssh_port, meta.ssh_key, meta.ssh_user
113    );
114    Ok(())
115}
