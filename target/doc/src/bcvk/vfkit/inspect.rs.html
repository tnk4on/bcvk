1//! vm inspect — Show detailed VM information.
2
3use super::VmMetadata;
4use color_eyre::Result;
5
6/// Display detailed metadata for the named VM.
7pub fn run(name: &str, json: bool) -> Result<()> {
8    let meta = VmMetadata::load(name)?;
9
10    if json {
11        println!("{}", serde_json::to_string_pretty(&meta)?);
12        return Ok(());
13    }
14
15    let state = if meta.is_alive() {
16        "running"
17    } else {
18        "stopped"
19    };
20
21    println!("Name:       {}", meta.name);
22    println!("State:      {}", state);
23    println!("Disk:       {}", meta.disk_image);
24    println!("CPUs:       {}", meta.cpus);
25    println!("Memory:     {} MiB", meta.memory);
26    println!("GUI:        {}", meta.gui);
27    println!("Created:    {}", meta.created);
28    println!();
29    println!("Processes:");
30    if meta.vfkit_pid > 0 {
31        println!(
32            "  vfkit:    PID {} ({})",
33            meta.vfkit_pid,
34            if meta.is_alive() {
35                "running"
36            } else {
37                "stopped"
38            }
39        );
40    }
41    if meta.gvproxy_pid > 0 {
42        println!("  gvproxy:  PID {}", meta.gvproxy_pid);
43    }
44    println!();
45    println!("SSH:");
46    println!("  Port:     {}", meta.ssh_port);
47    println!("  User:     {}", meta.ssh_user);
48    println!("  Key:      {}", meta.ssh_key);
49    if state == "running" {
50        println!();
51        println!(
52            "  ssh -p {} -i {} {}@localhost",
53            meta.ssh_port, meta.ssh_key, meta.ssh_user
54        );
55    }
56    println!();
57    println!("Files:");
58    println!("  EFI store:  {}", meta.efi_store);
59    println!("  Serial log: {}", meta.serial_log);
60
61    Ok(())
62}
