21/// Subcommands for persistent VM management via vfkit.
22#[derive(Debug, Subcommand)]
23pub enum VmCommands {
24    /// Run a persistent VM from a disk image
25    Run(run::VmRunOpts),
26
27    /// List all persistent VMs
28    #[clap(name = "list", alias = "ls")]
29    List {
30        /// Output in JSON format
31        #[clap(long)]
32        json: bool,
33    },
34
35    /// SSH into a running VM
36    Ssh(ssh::VmSshOpts),
37
38    /// Stop a running VM
39    Stop {
40        /// VM name
41        name: String,
42    },
43
44    /// Start a stopped VM
45    Start(start::VmStartOpts),
46
47    /// Remove a VM and its metadata
48    #[clap(name = "rm")]
49    Remove(rm::VmRmOpts),
50
51    /// Remove all VMs
52    #[clap(name = "rm-all")]
53    RemoveAll {
54        /// Force removal without confirmation
55        #[clap(short, long)]
56        force: bool,
57    },
58
59    /// Show detailed VM information
60    Inspect {
61        /// VM name
62        name: String,
63        /// Output in JSON format
64        #[clap(long)]
65        json: bool,
66    },
67}
68
69impl VmCommands {
70    /// Dispatch to the appropriate subcommand handler.
71    pub fn run(self) -> Result<()> {
72        match self {
73            VmCommands::Run(opts) => run::run(opts),
74            VmCommands::List { json } => list::run(json),
75            VmCommands::Ssh(opts) => ssh::run(opts),
76            VmCommands::Stop { name } => stop::run(&name),
77            VmCommands::Start(opts) => start::run(opts),
78            VmCommands::Remove(opts) => rm::run(opts),
79            VmCommands::RemoveAll { force } => rm_all::run(force),
80            VmCommands::Inspect { name, json } => inspect::run(&name, json),
81        }
82    }
83}

87/// Persistent VM metadata, stored as JSON in `~/.local/share/bcvk/vms/`.
88#[derive(serde::Serialize, serde::Deserialize, Debug, Clone)]
89pub struct VmMetadata {
90    /// VM name used as identifier.
91    pub name: String,
92    /// Path to the disk image file.
93    pub disk_image: String,
94    /// PID of the vfkit process.
95    pub vfkit_pid: u32,
96    /// PID of the gvproxy network proxy process.
97    pub gvproxy_pid: u32,
98    /// Host-side SSH port forwarded to the VM.
99    pub ssh_port: u16,
100    /// Path to the SSH private key.
101    pub ssh_key: String,
102    /// SSH username for connecting to the VM.
103    pub ssh_user: String,
104    /// Number of vCPUs allocated.
105    pub cpus: u32,
106    /// Memory in megabytes.
107    pub memory: u32,
108    /// Path to the EFI variable store file.
109    pub efi_store: String,
110    /// Path to the serial console log file.
111    pub serial_log: String,
112    /// Whether GUI mode is enabled.
113    pub gui: bool,
114    /// ISO 8601 timestamp when the VM was created.
115    pub created: String,
116    /// Current VM state (running, stopped).
117    pub state: String,
118}
119
120impl VmMetadata {
121    /// Return the directory path for persistent VM metadata files.
122    pub fn vms_dir() -> PathBuf {
123        dirs::home_dir()
124            .expect("cannot determine home directory")
125            .join(".local/share/bcvk/vms")
126    }
127
128    /// Save metadata to a JSON file in the VMs directory.
129    pub fn save(&self) -> Result<()> {
130        let dir = Self::vms_dir();
131        fs::create_dir_all(&dir)?;
132        let path = dir.join(format!("{}.json", self.name));
133        fs::write(&path, serde_json::to_string_pretty(self)?)?;
134        Ok(())
135    }
136
137    /// Load metadata for the named VM from its JSON file.
138    pub fn load(name: &str) -> Result<Self> {
139        let path = Self::vms_dir().join(format!("{}.json", name));
140        let data = fs::read_to_string(&path)?;
141        Ok(serde_json::from_str(&data)?)
142    }
143
144    /// Remove metadata file for the named VM.
145    pub fn remove(name: &str) {
146        let path = Self::vms_dir().join(format!("{}.json", name));
147        let _ = fs::remove_file(path);
148    }
149
150    /// List all persistent VM metadata from the VMs directory.
151    pub fn list_all() -> Result<Vec<Self>> {
152        let dir = Self::vms_dir();
153        if !dir.exists() {
154            return Ok(Vec::new());
155        }
156        let mut vms = Vec::new();
157        for entry in fs::read_dir(&dir)? {
158            let path = entry?.path();
159            if path.extension().and_then(|e| e.to_str()) != Some("json") {
160                continue;
161            }
162            if let Ok(data) = fs::read_to_string(&path) {
163                if let Ok(meta) = serde_json::from_str::<Self>(&data) {
164                    vms.push(meta);
165                }
166            }
167        }
168        Ok(vms)
169    }
170
171    /// Check if the VM process is still alive via kill(pid, 0).
172    pub fn is_alive(&self) -> bool {
173        if self.vfkit_pid == 0 {
174            return false;
175        }
176        rustix::process::test_kill_process(
177            rustix::process::Pid::from_raw(self.vfkit_pid as i32).unwrap(),
178        )
179        .is_ok()
180    }
181}

184mod tests {
185    use super::*;

187    fn sample_vm_metadata(name: &str) -> VmMetadata {
188        VmMetadata {
189            name: name.to_string(),
190            disk_image: "/tmp/disk.raw".to_string(),
191            vfkit_pid: 0,
192            gvproxy_pid: 0,
193            ssh_port: 2222,
194            ssh_key: "/tmp/key".to_string(),
195            ssh_user: "root".to_string(),
196            cpus: 2,
197            memory: 4096,
198            efi_store: "/tmp/efi.fd".to_string(),
199            serial_log: "/tmp/serial.log".to_string(),
200            gui: false,
201            created: "2026-01-01T00:00:00Z".to_string(),
202            state: "running".to_string(),
203        }
204    }
205
206    #[test]
207    fn test_vm_metadata_roundtrip() {
208        let meta = sample_vm_metadata("test-vm");
209        let json = serde_json::to_string_pretty(&meta).unwrap();
210        let loaded: VmMetadata = serde_json::from_str(&json).unwrap();
211        assert_eq!(loaded.name, "test-vm");
212        assert_eq!(loaded.disk_image, "/tmp/disk.raw");
213        assert_eq!(loaded.cpus, 2);
214        assert_eq!(loaded.memory, 4096);
215        assert_eq!(loaded.ssh_user, "root");
216        assert_eq!(loaded.state, "running");
217        assert!(!loaded.gui);
218    }
219
220    #[test]
221    fn test_vm_metadata_save_load_remove() {
222        let dir = tempfile::tempdir().unwrap();
223        let json_path = dir.path().join("myvm.json");
224        let meta = sample_vm_metadata("myvm");
225        fs::write(&json_path, serde_json::to_string_pretty(&meta).unwrap()).unwrap();
226        let data = fs::read_to_string(&json_path).unwrap();
227        let loaded: VmMetadata = serde_json::from_str(&data).unwrap();
228        assert_eq!(loaded.name, "myvm");
229        assert_eq!(loaded.ssh_port, 2222);
230        fs::remove_file(&json_path).unwrap();
231        assert!(!json_path.exists());
232    }
233
234    #[test]
235    fn test_vm_metadata_list_from_dir() {
236        let dir = tempfile::tempdir().unwrap();
237        for i in 0..3 {
238            let meta = sample_vm_metadata(&format!("vm-{i}"));
239            let path = dir.path().join(format!("vm-{i}.json"));
240            fs::write(&path, serde_json::to_string(&meta).unwrap()).unwrap();
241        }
242        fs::write(dir.path().join("notes.txt"), "ignored").unwrap();
243
244        let mut vms = Vec::new();
245        for entry in fs::read_dir(dir.path()).unwrap() {
246            let path = entry.unwrap().path();
247            if path.extension().and_then(|e| e.to_str()) != Some("json") {
248                continue;
249            }
250            if let Ok(data) = fs::read_to_string(&path) {
251                if let Ok(meta) = serde_json::from_str::<VmMetadata>(&data) {
252                    vms.push(meta);
253                }
254            }
255        }
256        assert_eq!(vms.len(), 3);
257        let mut names: Vec<_> = vms.iter().map(|v| v.name.clone()).collect();
258        names.sort();
259        assert_eq!(names, vec!["vm-0", "vm-1", "vm-2"]);
260    }
261
262    #[test]
263    fn test_vm_metadata_is_alive_zero_pid() {
264        let meta = sample_vm_metadata("dead-vm");
265        assert!(!meta.is_alive());
266    }
